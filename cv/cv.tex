\section{Research Profile}
Computational geodynamicist studying how mantle flow, subduction, lithospheric deformation, and mantle rheology shape observable surface motions. I build physics-based numerical models and compare model predictions with geological and geophysical constraints, with current emphasis on Western U.S. vertical motion and Cascadia.

\section{Research Experience}
\cventry{Western U.S. Vertical Motion — UT Austin}{Aug. 2024--Present}
{Graduate Researcher, UT Institute for Geophysics, Advisor: Thorsten Becker}{Austin, TX}
\begin{itemize}
  \item Build global and regional geodynamic models to quantify how subduction, mantle flow, and rheology influence vertical crustal motion and dynamic topography.
  \item Develop Cascadia subduction-zone model configurations and compare model predictions with geological/geophysical observations.
  \item Design numerical experiments to test sensitivity to slab geometry, mantle rheology, density structure, and boundary conditions.
  \item Run and analyze large-scale simulations using \href{https://aspect.geodynamics.org/}{ASPECT} and high-performance computing workflows; process model output with Python, GMT, and related tools.
\end{itemize}

\cventry{Western Pacific double-subduction drives Gulf of California rifting — UT Austin}{Feb. 2025--Jul. 2026}
{Graduate Researcher, UT Institute for Geophysics, Advisor: Thorsten Becker}{Austin, TX}
\begin{itemize}
\item Set up global reference geodynamic model to evaluate velocity field and stress state of lithosphere.
\item Compare double and single subduction configurations, evaluate the additional extensional stress (~15 MPa) imparted on Gulf of California rifting from western Pacific double-subduction system.
\end{itemize}

\cventry{3D Regional Model: Intraplate Deformation within East Asia (remote) — UIUC}{May. 2023--Oct. 2023}
{Visiting Student, Department of Earth Science and Environmental Change, Advisor: Lijun Liu}{Urbana, IL}
\begin{itemize}
  \item Conducted time-dependent 3-D regional geodynamic modeling of intraplate deformation in East Asia.
  \item Analyse the velocity field and stress field near Tan-Lu Fault and investigate how its strike-slip pattern changes through time.
\end{itemize}

\cventry{Rock Dynamics Simulation of Planetary Crust Formation — USTC}{Jan. 2023--Aug. 2023}
{Undergraduate Researcher, School of Earth and Space Sciences, Advisor: Wei Leng}{Hefei, China}
\begin{itemize}
  \item Set up a 3D regional model to evaluate the effect of hydrated plume on continental crust formation.
  \item Model shows that the generation of continental crust is almost impossible with a dry plume, which indicates the necessity of deep water's participation in this process. 
\end{itemize}

\cventry{Réunion plume and double-subduction coupling as a driver of India's two-stage rapid northward motion — USTC}{Apr. 2022--Jun. 2024}
{Undergraduate Researcher, School of Earth and Space Sciences, Advisor: Jinshui Huang}{Hefei, China}
\begin{itemize}
  \item Develop time-dependent geodynamic models coupling early Réunion plume-lithosphere interaction (~70-66 Ma) with an evolving double-subduction system, and quantify their respective contributions to India's acceleration, sustained rapid motion, and eventual slowdown.
\end{itemize}
